## M2 — Test Expansion

### Task 1: `tests/todoDomain.test.ts` — domain entity transform'ları

**Objective:** Saf domain mantığını test et — hiçbir mock gerektirmez, en yüksek değer/çaba oranı.

**Files:**
- Create: `tests/todoDomain.test.ts`

**Steps:**
1. Test dosyasını yaz:

```typescript
import { describe, it, expect } from "vitest";
import {
  createTodo,
  toggleTodo,
  updateTodoStatus,
  updateTodoPriorities,
} from "@/domain/entities/Todo.js";

describe("createTodo", () => {
  it("defaults: incomplete, todo status, no repeat, general category", () => {
    const todo = createTodo({ text: "Test task" });
    expect(todo.completed).toBe(false);
    expect(todo.status).toBe("todo");
    expect(todo.repeat).toBe("none");
    expect(todo.category).toBe("general");
    expect(todo.lastCompletedDate).toBeNull();
    expect(todo.dueDate).toBeUndefined();
  });
});

describe("toggleTodo", () => {
  it("toggles completed state and records date", () => {
    const todo = createTodo({ text: "A" });
    const done = toggleTodo(todo);
    expect(done.completed).toBe(true);
    expect(done.status).toBe("done");
    expect(done.lastCompletedDate).toBeTruthy();
    expect(done.completedDates).toHaveLength(1);
    // original immutable
    expect(todo.completed).toBe(false);
  });
});

describe("updateTodoStatus", () => {
  it("done → completed=true + completedDates append", () => {
    const todo = createTodo({ text: "A" });
    const done = updateTodoStatus(todo, "done");
    expect(done.completed).toBe(true);
    expect(done.completedDates).toHaveLength(1);
  });
});

describe("updateTodoPriorities", () => {
  it("sets urgent/important", () => {
    const todo = createTodo({ text: "A" });
    const prioritized = updateTodoPriorities(todo, true, true);
    expect(prioritized.urgent).toBe(true);
    expect(prioritized.important).toBe(true);
  });
});
```

2. **Verify:** `npx vitest run tests/todoDomain.test.ts` → 4-5 passed

### Task 2: `tests/todoUseCases.test.ts` — use case orchestration (mock repo)

**Objective:** `AddTodoUseCase` + `ToggleTodoUseCase` + `DeleteTodoUseCase`'in repo orchestration'ını mock'lu test et.

**Files:**
- Create: `tests/todoUseCases.test.ts`

**Steps:**
1. In-memory mock repo yaz (setup.ts'in `chrome.storage` mock'u üzerinden `ChromeStorageTodoRepository` bile kullanılabilir, ama izole mock daha hızlı):

```typescript
import { describe, it, expect, vi, beforeEach } from "vitest";
import { AddTodoUseCase } from "@/application/use-cases/todo/AddTodoUseCase.js";
import { ToggleTodoUseCase } from "@/application/use-cases/todo/ToggleTodoUseCase.js";
import { DeleteTodoUseCase } from "@/application/use-cases/todo/DeleteTodoUseCase.js";
import { createTodo } from "@/domain/entities/Todo.js";
import type { Todo } from "@/domain/entities/Todo.js";
import type { ITodoRepository } from "@/domain/repositories/ITodoRepository.js";
import type { ISyncRepository } from "@/domain/repositories/ISyncRepository.js";

function makeMockSyncRepo(): ISyncRepository {
  return {
    getSyncSettings: async () => ({
      enabled: false,
      tasksEnabled: false,
      calendarEnabled: false,
    }),
    setSyncSettings: async () => {},
  };
}

describe("AddTodoUseCase", () => {
  let todos: Todo[];
  let repo: ITodoRepository;
  let syncRepo: ISyncRepository;

  beforeEach(() => {
    todos = [];
    repo = {
      getAll: async () => [...todos],
      saveAll: async (t) => {
        todos = [...t];
      },
      getById: async (id) => todos.find((t) => t.id === id),
    };
    syncRepo = makeMockSyncRepo();
  });

  it("adds a todo via repo.saveAll", async () => {
    const uc = new AddTodoUseCase(repo, syncRepo);
    const { todo } = await uc.execute({ text: "New task", repeat: "none" });
    expect(todo.text).toBe("New task");
    expect(todo.completed).toBe(false);
    expect(todos).toHaveLength(1);
    expect(todos[0].text).toBe("New task");
  });
});
```

> **Not:** `syncPort` optional — `syncSettings.enabled=false` iken sync atlanır, `new AddTodoUseCase(repo, syncRepo)` iki argümanla kurulabilir (imza `syncPort?` optional). Toggle/Delete use case'lerini okuyup aynı desende test et (repo'da index bazlı mı id bazlı mı çalışıyorlar, imzalarını doğrula).

## Riskler & Notlar

1. **`ToggleTodoUseCase` / `DeleteTodoUseCase` imzaları:** `index` bazlı mı `id` bazlı mı — Task 2'de use case dosyasını oku, test'i gerçek imzaya göre yaz. Plan'daki örnek `AddTodoUseCase` için doğrulanmış.
2. **`tsconfig` test kapsamı:** `tsc --noEmit` test dosyalarını da derliyorsa, vitest import'ları (`describe` vb.) tip hatası verebilir — `tests/setup.ts` zaten derleniyor (mevcut testler tsc'den geçiyor), yani sorun olmamalı.
3. **Mock sync repo:** `getSyncSettings` dönüşü `GoogleSyncSettings` tipine uymalı (`enabled`, `tasksEnabled`, `calendarEnabled` zorunlu). `userEmail` optional.
4. **Kapsam disiplini:** Test-only. `src/` dosyalarına dokunma — test'ler üretim kodunu değiştirmeden yazılabilir olmalı.
5. **Mevcut `USER_DATA_WRITERS` listesi** (`cloudBackup.test.ts:17-51`) kapsam genişletmesiyle çelişmez — yalnızca statik string kontrolü yapıyor, yeni test dosyaları listeye girmez.
